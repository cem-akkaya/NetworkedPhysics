+void FSuspensionSimModule::GetWorldRaycastLocation(const FTransform& BodyTransform, float WheelRadius, float WheelWidth, float WheelLeanThreshold, FSpringTrace& OutTrace)
+       {
+               FVector LocalDirection = Setup().SuspensionAxis;
+               FVector LocalLocation = GetParentRelativeTransform().GetLocation();
+
+               FVector WorldRight = BodyTransform.GetUnitAxis(EAxis::Y);
+               FVector GlobalUp = FVector(0.f, 0.f, 1.f);
+
+               // Find the lean direction relative left or right
+               float Dot = FVector::DotProduct(WorldRight, GlobalUp);
+               float AngleRadians = FMath::Asin(Dot);
+
+               // Different tire profiles and behaviors mean the contact point shift may need to reach its maximum before the vehicle is fully on its side. Mapping the lean angle to this factor.
+               // MaxEffectAngleDegrees allows for tuning the aggressiveness of the offset based on the specific wheel/tire setup.
+               // If there is no wheel, or the width of the wheel is 0, the lean factor will be 0 thus it will behave as a static vertical only spring.
+               float LeanFactor = FMath::Clamp(FMath::RadiansToDegrees(AngleRadians) / WheelLeanThreshold, -1.0f, 1.0f) * -1.0f;
+
+               float SideShift = LeanFactor * (WheelWidth * 0.5f);
+
+
+               // Apply the lean offset to the world
+               FVector WorldLocation = BodyTransform.TransformPosition(LocalLocation);
+               WorldLocation += WorldRight * SideShift;
+
+               // Do trace
+               FVector WorldDirection = BodyTransform.TransformVector(LocalDirection);
+               OutTrace.Start = WorldLocation - WorldDirection * (Setup().MaxRaise);
+               OutTrace.End = WorldLocation + WorldDirection * (Setup().MaxDrop + WheelRadius);
+       }
